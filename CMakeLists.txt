cmake_minimum_required(VERSION 3.16)
project(EXCALIBUR_LOTTERY369 VERSION 1.0.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(EXLOTTERY_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(EXLOTTERY_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
    add_link_options(-fsanitize=address,undefined)
endif()
# Opt-in tuning for the machine that builds it. Yields a faster but non-portable
# binary (may use AVX/SSE4 anywhere), so it stays off for the distributed build.
# LTO was measured slower on the EC hot loop here and is deliberately not enabled.
option(EXLOTTERY_NATIVE "Tune for the host CPU (-march=native); non-portable binary" OFF)
if(EXLOTTERY_NATIVE)
    include(CheckCXXCompilerFlag)
    check_cxx_compiler_flag("-march=native" EXLOTTERY_HAS_NATIVE)
    if(EXLOTTERY_HAS_NATIVE)
        add_compile_options(-march=native)
    else()
        message(WARNING "Compilador rejeitou -march=native; mantendo flags padrão")
    endif()
endif()
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
find_path(BOOST_INCLUDE_DIR boost/multiprecision/cpp_int.hpp)
if(NOT BOOST_INCLUDE_DIR)
    message(FATAL_ERROR "Install Boost development headers (libboost-dev)")
endif()
include(cmake/Crypto.cmake)
add_library(exlottery_core STATIC src/uint256.cpp src/puzzle_range.cpp src/allowlist.cpp
    src/mixed_radix.cpp src/scheduler.cpp src/checkpoint.cpp src/engine.cpp
    src/catalog.cpp src/public_puzzles.cpp src/presentation.cpp)
target_include_directories(exlottery_core PUBLIC include "${BOOST_INCLUDE_DIR}")
target_compile_options(exlottery_core PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(exlottery_core PUBLIC exlottery_crypto OpenSSL::Crypto Threads::Threads)
add_executable(exlottery src/main.cpp)
target_link_libraries(exlottery PRIVATE exlottery_core)
configure_file(panel/app.py panel/app.py COPYONLY)
foreach(list_name primeiro compartilhada)
    configure_file(grupos/${list_name}.txt grupos/${list_name}.txt COPYONLY)
endforeach()
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    find_program(XVFB_RUN xvfb-run)
    find_program(XAUTH xauth)
    execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import tkinter"
                    RESULT_VARIABLE TK_RESULT OUTPUT_QUIET ERROR_QUIET)
    if(XVFB_RUN AND XAUTH AND TK_RESULT EQUAL 0)
        add_test(NAME panel COMMAND "${XVFB_RUN}" -a "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/panel.py" $<TARGET_FILE:exlottery>)
    endif()
    foreach(suite math crypto scheduler)
        add_executable(test_${suite} tests/${suite}.cpp)
        target_link_libraries(test_${suite} PRIVATE exlottery_core)
        target_compile_options(test_${suite} PRIVATE -UNDEBUG)
        add_test(NAME ${suite} COMMAND test_${suite})
    endforeach()
    add_test(NAME public_catalog COMMAND ${Python3_EXECUTABLE}
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/public_catalog.py" $<TARGET_FILE:exlottery>)
    add_test(NAME panel_run COMMAND ${Python3_EXECUTABLE}
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/panel_run.py" $<TARGET_FILE:exlottery>)
    foreach(suite cli dry_run_all fixture160 restart stop_on_hit search)
        add_test(NAME ${suite} COMMAND ${Python3_EXECUTABLE}
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.py" $<TARGET_FILE:exlottery> ${suite})
    endforeach()
endif()
